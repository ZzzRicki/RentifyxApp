"use client";

import React, { useState, useEffect } from "react";
import axios from "axios";
import VehicleTable from "@/features/vehicles/VehicleTable";
import VehicleModal from "@/features/vehicles/RegisterVehicleModal";

interface Vehicle {
  id: number;
  brand: string;
  model: string;
  year: number;
  transmissionType: number;
  passengerCapacity: number;
  dailyPrice: number;
  status: number;
  location: string;
  imageUrls: string[];
}

const VehiclesPage = () => {
  const [vehicles, setVehicles] = useState<Vehicle[]>([]);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState<string | null>(null);
  const [showModal, setShowModal] = useState(false); // Modal visibility state
  const [newVehicle, setNewVehicle] = useState({
    brand: "",
    model: "",
    year: 0,
    dailyPrice: 0,
    status: 0,
    transmissionType: 0,
    passengerCapacity: 0,
    location: "",
    images: null as File[] | null,
  });

  useEffect(() => {
    const fetchVehicles = async () => {
      try {
        const response = await axios.get("http://localhost:5267/api/Vehicles");
        if (response.data.success) {
          setVehicles(response.data.data);
        } else {
          setError("Failed to fetch vehicles");
        }
        // eslint-disable-next-line @typescript-eslint/no-unused-vars
      } catch (error) {
        setError("Error fetching vehicles");
      } finally {
        setLoading(false);
      }
    };

    fetchVehicles();
  }, []);

  const handleAddVehicle = () => {
    setShowModal(true);
  };

  const handleCloseModal = () => {
    setShowModal(false);
  };

  const handleChange = (e: React.ChangeEvent<HTMLInputElement>) => {
    setNewVehicle({
      ...newVehicle,
      [e.target.name]: e.target.value,
    });
  };

  const handleFileChange = (files: File[]) => {
    setNewVehicle((prevState) => ({
      ...prevState,
      images: files,
    }));
  };

  const handleSelectChange = (e: React.ChangeEvent<HTMLSelectElement>) => {
    setNewVehicle({
      ...newVehicle,
      [e.target.name]: e.target.value,
    });
  };

  const handleSubmit = async (e: React.FormEvent) => {
    e.preventDefault();

    const formData = new FormData();
    formData.append("Brand", newVehicle.brand);
    formData.append("Model", newVehicle.model);
    formData.append("Year", newVehicle.year.toString());
    formData.append("TransmissionType", newVehicle.transmissionType.toString());
    formData.append(
      "PassengerCapacity",
      newVehicle.passengerCapacity.toString()
    );
    formData.append("DailyPrice", newVehicle.dailyPrice.toString());
    formData.append("Status", newVehicle.status.toString());
    formData.append("Location", newVehicle.location);

    // Append multiple image files to the form data
    if (newVehicle.images) {
      for (let i = 0; i < newVehicle.images.length; i++) {
        formData.append("Images", newVehicle.images[i]);
      }
    }

    try {
      const response = await fetch("http://localhost:5267/api/Vehicles", {
        method: "POST",
        headers: {
          Authorization: `Bearer ${""}`, // Replace `yourToken` with actual token
          accept: "*/*",
        },
        body: formData, // Set the formData as the body
      });

      if (response.ok) {
      } else {
        const errorData = await response.json();
        console.error("Error:", errorData);
      }
    } catch (error) {
      console.error("Error adding vehicle:", error);
    }
  };

  if (loading) return <p>Cargando vehículos...</p>;
  if (error) return <p>Error: {error}</p>;

  return (
    <div>
      <div className="flex justify-between items-center mb-6">
        <h1 className="text-2xl font-bold text-[#171717]">
          Gestionar Vehículos
        </h1>
        <button
          onClick={handleAddVehicle}
          className="bg-green-600 hover:bg-[#238636] text-white font-bold py-2 px-4 rounded-md"
        >
          Agregar Vehículo
        </button>
      </div>

      <VehicleTable vehicles={vehicles} />

      <VehicleModal
        isOpen={showModal}
        onClose={handleCloseModal}
        newVehicle={newVehicle}
        handleSelectChange={handleSelectChange}
        handleChange={handleChange}
        handleFileChange={handleFileChange}
        handleSubmit={handleSubmit}
      />
    </div>
  );
};

export default VehiclesPage;
